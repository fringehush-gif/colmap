cmake_minimum_required(VERSION 3.18)
project(FindOnnxruntimeTest LANGUAGES NONE)

set(CMAKE_FIND_USE_CMAKE_SYSTEM_PATH FALSE)
set(CMAKE_FIND_USE_SYSTEM_ENVIRONMENT_PATH FALSE)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY FALSE)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY FALSE)
set(CMAKE_FIND_LIBRARY_PREFIXES "")
set(CMAKE_FIND_LIBRARY_SUFFIXES ".lib")
set(fake "${CMAKE_BINARY_DIR}/fake")
set(CMAKE_FIND_ROOT_PATH "${fake}")
set(CMAKE_FIND_ROOT_PATH_MODE_PACKAGE ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)
file(MAKE_DIRECTORY "${fake}/include/onnxruntime" "${fake}/lib" "${fake}/bin")
file(WRITE "${fake}/include/onnxruntime/onnxruntime_cxx_api.h" "")
foreach(lib onnxruntime onnxruntime_providers_shared onnxruntime_providers_cuda)
    file(WRITE "${fake}/lib/${lib}.lib" "fixture import ${lib}")
    file(WRITE "${fake}/bin/${lib}.dll" "fixture runtime ${lib}")
endforeach()
set(onnxruntime_INCLUDE_DIR_HINTS "${fake}/include" CACHE PATH "" FORCE)
set(onnxruntime_LIBRARY_DIR_HINTS "${fake}/lib" CACHE PATH "" FORCE)
# Exercise Windows DLL staging on each host.
set(IS_WINDOWS TRUE)
set(VCPKG_INSTALLED_DIR "${CMAKE_BINARY_DIR}/installed-${CUDA_ENABLED}")
set(VCPKG_TARGET_TRIPLET "test")
set(staged "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/bin")
file(REMOVE "${staged}/onnxruntime.dll"
            "${staged}/onnxruntime_providers_shared.dll"
            "${staged}/onnxruntime_providers_cuda.dll")
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../..")
find_package(onnxruntime MODULE REQUIRED)

get_target_property(link_libraries onnxruntime::onnxruntime INTERFACE_LINK_LIBRARIES)
if(NOT link_libraries STREQUAL "${fake}/lib/onnxruntime.lib")
    message(FATAL_ERROR "Fallback target links provider libraries: ${link_libraries}")
endif()
foreach(lib onnxruntime onnxruntime_providers_shared)
    if(NOT EXISTS "${staged}/${lib}.dll")
        message(FATAL_ERROR "Missing staged runtime: ${lib}")
    endif()
endforeach()
if(CUDA_ENABLED AND NOT EXISTS "${staged}/onnxruntime_providers_cuda.dll")
    message(FATAL_ERROR "Missing staged CUDA runtime")
elseif(NOT CUDA_ENABLED AND EXISTS "${staged}/onnxruntime_providers_cuda.dll")
    message(FATAL_ERROR "CUDA-off configure staged cached CUDA provider")
endif()
